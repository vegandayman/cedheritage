const BASIC_LANDS = ['plains', 'island', 'swamp', 'mountain', 'forest', 'snow-covered plains', 'snow-covered island', 'snow-covered swamp', 'snow-covered mountain', 'snow-covered forest']; let heritageDatabase = []; let isDbLoaded = false; async function loadDatabase() { const btn = document.getElementById('loadDbBtn'); const status = document.getElementById('dbStatus'); btn.disabled = true; btn.innerText = "LOADING..."; status.innerHTML = "Fetching local database file..."; try { const response = await fetch('./heritage_cards.json'); if (!response.ok) throw new Error('Database file not found. Ensure heritage_cards.json is uploaded.'); const jsonPayload = await response.json(); if (Array.isArray(jsonPayload)) { heritageDatabase = jsonPayload; } else { heritageDatabase = jsonPayload.cards || []; if (jsonPayload.lastUpdated) { window.dbLastUpdated = new Date(jsonPayload.lastUpdated).toLocaleString(); } } isDbLoaded = true; btn.style.display = 'none'; let dateString = window.dbLastUpdated ? ` (Updated: ${window.dbLastUpdated})` : ''; status.innerHTML = `Database Loaded: ${heritageDatabase.length} entries ready${dateString}.`; } catch (error) { console.error("Failed to load heritage_cards.json:", error); status.innerHTML = `Error: ${error.message}`; btn.disabled = false; btn.innerText = "RETRY LOAD DATABASE"; } } function searchCard() { const resultsDiv = document.getElementById('searchResults'); if (!isDbLoaded) { resultsDiv.innerHTML = 'You must load the database first.'; return; } const nameInput = document.getElementById('searchInput').value.trim().toLowerCase(); const selectedType = document.getElementById('typeSelect').value.toLowerCase(); const oracleInput = document.getElementById('oracleInput').value.trim().toLowerCase(); const matches = heritageDatabase.filter(card => { // Check if name matches base name OR any of the flavor names in the array const matchesName = nameInput === '' || card.n.includes(nameInput) || (card.f && card.f.some(flavor => flavor.includes(nameInput))); const matchesType = selectedType === '' || card.t.includes(selectedType); const matchesOracle = oracleInput === '' || (card.o && card.o.includes(oracleInput)); return matchesName && matchesType && matchesOracle; }); if (matches.length === 0) { resultsDiv.innerHTML = `No legal cards found matching your criteria.`; return; } const displayLimit = 50; let html = `

Found ${matches.length} matching entry/entries. ${matches.length > displayLimit ? `Displaying first ${displayLimit}.` : ''}

`; matches.slice(0, displayLimit).forEach(card => { if (card.i) { html += ``; } else { const displayName = card.n.replace(/\b\w/g, l => l.toUpperCase()); html += `
${displayName}
`; } }); html += '
'; resultsDiv.innerHTML = html; } function checkDeck() { const resultsDiv = document.getElementById('deckResults'); if (!isDbLoaded) { resultsDiv.innerHTML = 'You must load the database first.'; return; } const fileInput = document.getElementById('deckFile'); if (!fileInput.files[0]) { resultsDiv.innerHTML = 'Please select a file first.'; return; } const reader = new FileReader(); reader.onload = function(e) { const text = e.target.result; const lines = text.split('\n'); const cardNames = []; lines.forEach(line => { const cleanLine = line.trim(); if (!cleanLine || cleanLine.toLowerCase().startsWith('sideboard') || cleanLine.toLowerCase().startsWith('commander')) return; const match = cleanLine.match(/^(\d+x?\s+)?([^<\[\(\*]+)/); if (match && match[2]) { const name = match[2].trim(); if (name && !BASIC_LANDS.includes(name.toLowerCase())) { cardNames.push(name); } } }); if (cardNames.length === 0) { resultsDiv.innerHTML = 'No valid cards found to check (or deck is only basic lands).'; return; } let legalCount = 0; let illegalCount = 0; let listHTML = ''; cardNames.forEach(name => { const lowerName = name.toLowerCase(); const matchingCard = heritageDatabase.find(card => { const parts = card.n.split(' // '); // Check base name, split name, or if lowerName exists in the flavor array return card.n === lowerName || parts.includes(lowerName) || (card.f && card.f.includes(lowerName)); }); if (matchingCard) { legalCount++; let displayName = name; // Append the base name in parentheses if a flavor name was triggered if (matchingCard.f && matchingCard.f.includes(lowerName)) { const baseNameFormatted = matchingCard.n.replace(/\b\w/g, char => char.toUpperCase()); displayName = `${name} (${baseNameFormatted})`; } listHTML += `
✓ LEGAL: ${displayName}
`; } else { illegalCount++; const searchUrl = `https://scryfall.com/search?q=!%22${encodeURIComponent(name)}%22`; listHTML += `
✗ ILLEGAL: ${name}
`; } }); let summaryHTML = `

Legality Report

`; summaryHTML += `

Total Checked: ${cardNames.length}  |  Legal: ${legalCount}  |  Illegal: ${illegalCount}

`; resultsDiv.innerHTML = summaryHTML + listHTML; if (illegalCount === 0) { resultsDiv.innerHTML += '

All non-basic cards are Heritage Legal!

'; } }; reader.readAsText(fileInput.files[0]); }
